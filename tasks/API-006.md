**★ Q9 (C 절충) 4 디테일 정확 (sentry-error.ts)**:

```typescript
// ★ 디테일 ① import + 실 함수 호출 (dead code 0)
import * as Sentry from '@sentry/nextjs';

export function reportErrorToSentry(error: AppErrorDTO): void {
  // ★ 디테일 ③ console.error fallback (dev/test 환경)
  if (process.env.NODE_ENV !== 'production') {
    console.error('[AppError]', error.code, error.message, error.originalError);
  }
  // ★ 디테일 ④ runtime no-op 안전성 (Sentry SDK 공식 보장)
  Sentry.withScope((scope) => {
    scope.setLevel(error.sentryLevel ?? 'error');
    scope.setContext('appError', { code: error.code, httpStatus: error.httpStatus });
    if (error.originalError !== undefined) {
      Sentry.captureException(error.originalError);
    } else {
      Sentry.captureMessage(error.message);
    }
  });
}

// ★ 디테일 ② Sentry.init() = MON-001 위임 주석 (모듈 상단 주석)
// Sentry.init({ dsn: NEXT_PUBLIC_SENTRY_DSN }) is deferred to MON-001 (Wave 5 트랙 P).
// This module assumes Sentry is initialized elsewhere.
```
