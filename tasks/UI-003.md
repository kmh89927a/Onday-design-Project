### ★★ Kakao Maps SSR + next/dynamic 패턴 §

```tsx
// map-canvas.tsx
const MapCanvasKakao = dynamic(() => import("./map-canvas-kakao"), {
  ssr: false,
});

// map-canvas-kakao.tsx
"use client";
import Script from "next/script";
import { Map, MapMarker } from "react-kakao-maps-sdk";

<Script
  src={`//dapi.kakao.com/v2/maps/sdk.js?autoload=false&appkey=${appKey}`}
  strategy="afterInteractive"
  onLoad={() => window.kakao.maps.load(() => setReady(true))}
/>
```

- SSR 안전 (★ window 의존 SDK = ssr:false 박힘)
